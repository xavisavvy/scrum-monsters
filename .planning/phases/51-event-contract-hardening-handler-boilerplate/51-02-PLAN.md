  export function registerSyncedLobbyHandler<E extends keyof ServerToClientEvents>(
    socket: TypedClientSocket, event: E,
    updater: (data: Parameters<ServerToClientEvents[E]>[0], lobby: Lobby) => Partial<Lobby> | null
  ): void;   // routes handleEvent seq-guard → if processed → null-checks currentLobby → setLobby({ ...currentLobby, ...update }) when update !== null
  export function registerSyncedHandler<E extends keyof ServerToClientEvents>(
    socket: TypedClientSocket, event: E,
    handler: (data: Parameters<ServerToClientEvents[E]>[0]) => void
  ): void;   // routes handleEvent seq-guard → if processed → handler(data) (handler owns ALL store actions)
  export function teardownSyncedHandlers(socket: TypedClientSocket): void;  // loops _registeredEvents, socket.off each, clears array
  const _registeredEvents: Array<keyof ServerToClientEvents> = [];  // pushed in both register fns

  Generic-inference note (A2): use `data as Parameters<ServerToClientEvents[E]>[0]` casts inside the callbacks if tsc cannot narrow E.
