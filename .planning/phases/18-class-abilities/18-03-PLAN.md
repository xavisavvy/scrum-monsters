**Zustand store:**
```typescript
interface CooldownState {
  abilityId: string;
  startedAt: number;  // Client timestamp when received
  expiresAt: number;  // Server-sent expiration timestamp
  durationMs: number; // Total cooldown duration
}

interface AbilitiesState {
  cooldowns: Map<string, CooldownState>; // abilityId -> CooldownState
  pendingAbility: string | null; // Optimistic UI: disable while request in flight

  // Actions
  startCooldown: (abilityId: string, durationMs: number, expiresAt: number) => void;
  isOnCooldown: (abilityId: string) => boolean;
  getCooldownProgress: (abilityId: string) => number; // 0-100 (0 = just started, 100 = ready)
  getRemainingSeconds: (abilityId: string) => number;
  clearCooldowns: () => void;
  setPendingAbility: (abilityId: string | null) => void;
  activateAbility: (abilityId: string) => void; // Sets pending + emits socket event
}
```

Key implementation details:
- `getCooldownProgress` calculates based on elapsed time vs durationMs: `(elapsed / durationMs) * 100`, clamped 0-100
- `getRemainingSeconds` calculates `Math.max(0, Math.ceil((expiresAt - Date.now()) / 1000))`
- `isOnCooldown` checks `Date.now() < expiresAt` AND accounts for 100ms client buffer (show ready slightly early per research pitfall #1)
- `activateAbility` sets pendingAbility to abilityId (optimistic disable), then the event listener clears it on response

**Server event sync hook (exported as useAbilitySync):**

```typescript
export function useAbilitySync() {
  const socket = useWebSocket(state => state.socket);
  const { startCooldown, clearCooldowns, setPendingAbility } = useAbilities();

  useEffect(() => {
    if (!socket) return;

    const handleCooldownStarted = (data: {
      playerId: string;
      abilityId: string;
      durationMs: number;
      expiresAt: number;
    }) => {
      // Only track own cooldowns (other players' cooldowns not needed for buttons)
      startCooldown(data.abilityId, data.durationMs, data.expiresAt);
      setPendingAbility(null); // Clear optimistic disable
    };

    const handleAbilityUsed = (data: { playerId: string; abilityId: string }) => {
      // Could be used for visual effects on other players' abilities
      // For now, just clear pending state if it's our ability
      setPendingAbility(null);
    };

    socket.on('ability:cooldown_started', handleCooldownStarted);
    socket.on('ability:used', handleAbilityUsed);

    return () => {
      socket.off('ability:cooldown_started', handleCooldownStarted);
      socket.off('ability:used', handleAbilityUsed);
    };
  }, [socket, startCooldown, clearCooldowns, setPendingAbility]);
}
```

**2. client/src/components/game/AbilityBar.tsx:**

Container that shows the current player's class abilities.

```tsx
export function AbilityBar() {
  const { currentPlayer, lobby } = useGameState();
  const { activateAbility, pendingAbility } = useAbilities();
  const { getMasteryData } = useClassMastery(); // from useClassMastery store

  // Only show during battle phase
  if (!lobby || lobby.gamePhase !== 'battle') return null;
  if (!currentPlayer?.avatar) return null;

  const playerClass = currentPlayer.avatar;
  const config = CLASS_ABILITY_CONFIGS[playerClass];
  if (!config) return null;

  // Get mastery tier for unlock checks
  const masteryData = getMasteryData(playerClass);
  const currentTier = masteryData?.currentTier ?? 'Novice';

  // Check which abilities are unlocked
  const isTierAtLeast = (required: MasteryTier): boolean => {
    const tiers: MasteryTier[] = ['Novice', 'Expert', 'Master'];
    return tiers.indexOf(currentTier) >= tiers.indexOf(required);
  };

  const handleActivate = (abilityId: string) => {
    activateAbility(abilityId);
  };

  return (
    <div className="flex gap-2 items-center">
      <AbilityButton
        ability={config.ability_1}
        isUnlocked={isTierAtLeast(config.ability_1.tier)}
        requiredTier={config.ability_1.tier}
        onActivate={handleActivate}
        disabled={pendingAbility !== null}
      />
      <AbilityButton
        ability={config.ability_2}
        isUnlocked={isTierAtLeast(config.ability_2.tier)}
        requiredTier={config.ability_2.tier}
        onActivate={handleActivate}
        disabled={pendingAbility !== null}
      />
    </div>
  );
}
```

Import CLASS_ABILITY_CONFIGS from '@shared/abilityTypes'. Import useGameState, useAbilities, useClassMastery from respective stores.
