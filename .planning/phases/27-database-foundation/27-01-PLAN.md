```typescript
constructor(connectionString: string, options?: {
  max?: number;
  idle_timeout?: number;
  connect_timeout?: number;
}) {
  this.sql = postgres(connectionString, {
    max: options?.max ?? 10,
    idle_timeout: options?.idle_timeout ?? 60,
    connect_timeout: options?.connect_timeout ?? 10,
    onnotice: () => {},  // Suppress NOTICE messages
  });
  this.db = drizzle(this.sql);
}
```

2. Add a private `sql` field to PgStorage to store the postgres.js instance (currently it's a local variable in constructor — promote it to class field):
- Change `private db: ReturnType<typeof drizzle>;` to also add `private sql: postgres.Sql;`
- The `sql` field is needed for both the close() method and the health check

3. Add a `close()` method to PgStorage:
```typescript
async close(): Promise<void> {
  await this.sql.end({ timeout: 5 });
}
```

4. Update `createStorage()` to pass pool options from env vars:
```typescript
function createStorage(): IStorage {
  if (process.env.DATABASE_URL) {
    console.log("Using PostgreSQL storage");
    return new PgStorage(process.env.DATABASE_URL, {
      max: parseInt(process.env.DB_POOL_MAX || "10", 10),
      idle_timeout: parseInt(process.env.DB_POOL_IDLE_TIMEOUT || "60", 10),
      connect_timeout: parseInt(process.env.DB_POOL_CONNECT_TIMEOUT || "10", 10),
    });
  }
  console.log("Using in-memory storage (no DATABASE_URL set)");
  return new MemStorage();
}
```

Note: createStorage() reads raw process.env because it runs at module load time (before validateEnv is called). The env vars have sensible defaults via parseInt fallbacks. This is intentional — validateEnv() in Plan 02 will validate before server.listen(), but storage creation happens at import time.
