    1. **Create `server/seoMiddleware.ts`:**
       A middleware function that injects Open Graph and Twitter card meta tags into the index.html template BEFORE sending it to the client. This runs for all requests (not just crawlers) so the meta tags are always in the initial HTML.

       The middleware reads the request URL path and maps it to meta tag values:
       ```typescript
       // server/seoMiddleware.ts

       const SITE_NAME = 'ScrumQuest';
       const SITE_URL = process.env.SITE_URL || 'https://scrumquest.app';
       const DEFAULT_OG_IMAGE = `${SITE_URL}/og-image.png`;

       interface MetaConfig {
         title: string;
         description: string;
         ogType?: string;
         ogImage?: string;
         canonicalUrl?: string;
       }

       // Mirror of client-side META_CONFIG -- keep in sync
       const ROUTE_META: Record<string, MetaConfig> = {
         '/': {
           title: 'ScrumQuest - Battle Tickets in Epic JRPG Style',
           description: 'Real-time multiplayer scrum poker estimation with JRPG-style boss battles. Turn sprint planning into an epic adventure!',
         },
         '/about': {
           title: 'About ScrumQuest - The Story Behind the Quest',
           description: 'Learn how ScrumQuest transforms boring sprint planning into exciting JRPG boss battles.',
         },
         '/features': {
           title: 'Features - ScrumQuest Game Mechanics & Tools',
           description: 'Explore ScrumQuest features: real-time voting, boss battles, team competitions, class abilities, and more.',
         },
         '/pricing': {
           title: 'Pricing - ScrumQuest Plans & Options',
           description: 'ScrumQuest pricing and plans. Free to play with optional features for teams.',
         },
         '/support': {
           title: 'Support - ScrumQuest Help & Contact',
           description: 'Get help with ScrumQuest. Find answers, report issues, or contact the team.',
         },
         '/play': {
           title: 'Play ScrumQuest - Start Your Quest',
           description: 'Create or join a ScrumQuest lobby. Estimate story points in epic JRPG style.',
         },
       };

       function getMetaForPath(path: string): MetaConfig {
         // Exact match first
         if (ROUTE_META[path]) return ROUTE_META[path];

         // Game route pattern
         if (path.startsWith('/game/')) {
           const lobbyId = path.split('/')[2];
           return {
             title: `ScrumQuest Battle - ${lobbyId}`,
             description: 'Live ScrumQuest estimation battle in progress. Join the fight!',
           };
         }

         // Default fallback
         return ROUTE_META['/'];
       }

       export function injectMetaTags(html: string, requestPath: string): string {
         const meta = getMetaForPath(requestPath);
         const canonicalUrl = `${SITE_URL}${requestPath === '/' ? '' : requestPath}`;
         const ogImage = meta.ogImage || DEFAULT_OG_IMAGE;

         const metaTags = `
           <title>${meta.title}</title>
           <meta name="description" content="${meta.description}" />
           <meta property="og:type" content="${meta.ogType || 'website'}" />
           <meta property="og:site_name" content="${SITE_NAME}" />
           <meta property="og:title" content="${meta.title}" />
           <meta property="og:description" content="${meta.description}" />
           <meta property="og:image" content="${ogImage}" />
           <meta property="og:url" content="${canonicalUrl}" />
           <meta name="twitter:card" content="summary_large_image" />
           <meta name="twitter:title" content="${meta.title}" />
           <meta name="twitter:description" content="${meta.description}" />
           <meta name="twitter:image" content="${ogImage}" />
           <link rel="canonical" href="${canonicalUrl}" />`;

         // Inject before </head> in the HTML template
         // Remove any existing <title> tag to prevent duplicates
         let processed = html.replace(/<title>.*?<\/title>/i, '');
         // Remove any existing meta description to prevent duplicates
         processed = processed.replace(/<meta\s+name="description"[^>]*>/i, '');
         // Inject new meta tags before </head>
         processed = processed.replace('</head>', `${metaTags}\n  </head>`);

         return processed;
       }
       ```

    2. **Integrate into server/vite.ts `setupVite` (dev mode):**
       In the dev-mode catch-all handler, call `injectMetaTags` on the HTML before sending:
       ```typescript
       import { injectMetaTags } from './seoMiddleware';

       // In the catch-all handler:
       let page = await vite.transformIndexHtml(url, template);
       page = injectMetaTags(page, req.originalUrl.split('?')[0]); // Strip query params
       res.status(200).set({ "Content-Type": "text/html" }).end(page);
       ```

    3. **Integrate into server/vite.ts `serveStatic` (production mode):**
       The production catch-all currently does `res.sendFile(path.resolve(distPath, "index.html"))`. Change this to read the file, inject meta tags, then send:
       ```typescript
       import { injectMetaTags } from './seoMiddleware';

       // In serveStatic:
       app.use("*", (req, res) => {
         const htmlPath = path.resolve(distPath, "index.html");
         const html = fs.readFileSync(htmlPath, 'utf-8');
         const processed = injectMetaTags(html, req.originalUrl.split('?')[0]);
         res.status(200).set({ "Content-Type": "text/html" }).end(processed);
       });
       ```

       For performance: cache the base HTML string on startup (read once, inject per-request). The meta tag injection is string replacement -- very fast.
