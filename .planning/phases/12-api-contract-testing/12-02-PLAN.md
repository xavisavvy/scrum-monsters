Create `shared/socket-schemas.ts` with Zod schemas for validating Socket.IO event payloads.

**Note:** Full zod-sockets integration requires refactoring socketHandlers.ts which is out of scope. This task creates validation schemas that can be used incrementally.

**Schema structure:**

```typescript
import { z } from 'zod';

// ============================================
// Core Types (matching gameEvents.ts)
// ============================================

export const PositionSchema = z.object({
  x: z.number(),
  y: z.number(),
});

export const TeamTypeSchema = z.enum(['developers', 'qa', 'spectators']);

export const AvatarClassSchema = z.enum([
  'ranger', 'rogue', 'bard', 'sorcerer', 'wizard',
  'warrior', 'paladin', 'cleric', 'oathbreaker', 'monk'
]);

export const GamePhaseSchema = z.enum([
  'lobby', 'avatar_selection', 'battle', 'scoring',
  'reveal', 'discussion', 'victory', 'next_level', 'game_over'
]);

// ============================================
// Client-to-Server Event Payloads
// ============================================

export const CreateLobbyPayloadSchema = z.object({
  lobbyName: z.string().min(1).max(100),
  hostName: z.string().min(1).max(50),
  initialSettings: z.object({
    timerSettings: z.object({
      enabled: z.boolean(),
      durationMinutes: z.number().positive(),
    }).optional(),
    jiraSettings: z.object({
      baseUrl: z.string().url().optional(),
    }).optional(),
    estimationSettings: z.object({
      scaleType: z.enum(['fibonacci', 'doubling', 'tshirt']),
      customTshirtMapping: z.record(z.string(), z.number()).optional(),
    }).optional(),
    customLobbyId: z.string().regex(/^[a-zA-Z0-9-]{3,30}$/).optional(),
  }).optional(),
});

export const JoinLobbyPayloadSchema = z.object({
  lobbyId: z.string().min(1),
  playerName: z.string().min(1).max(50),
});

export const SelectAvatarPayloadSchema = z.object({
  avatarClass: AvatarClassSchema,
});

export const SubmitScorePayloadSchema = z.object({
  score: z.union([z.number(), z.literal('?')]),
});

export const AttackBossPayloadSchema = z.object({
  damage: z.number().positive(),
});

export const PlayerPositionPayloadSchema = z.object({
  x: z.number(),
  y: z.number(),
});

export const LobbyPlayerPositionPayloadSchema = z.object({
  x: z.number(),
  y: z.number(),
  direction: z.string().optional(),
});

export const ReconnectWithTokenPayloadSchema = z.object({
  reconnectToken: z.string().min(1),
});

// ============================================
// Server-to-Client Event Payloads (for type inference)
// ============================================

export const LobbyCreatedPayloadSchema = z.object({
  lobby: z.object({
    id: z.string(),
    name: z.string(),
    hostId: z.string(),
    // ... (partial for type safety without full validation)
  }).passthrough(),
  inviteLink: z.string(),
});

export const GameErrorPayloadSchema = z.object({
  message: z.string(),
});

// ============================================
// Type Exports (infer from Zod schemas)
// ============================================

export type CreateLobbyPayload = z.infer<typeof CreateLobbyPayloadSchema>;
export type JoinLobbyPayload = z.infer<typeof JoinLobbyPayloadSchema>;
export type SelectAvatarPayload = z.infer<typeof SelectAvatarPayloadSchema>;
export type SubmitScorePayload = z.infer<typeof SubmitScorePayloadSchema>;
export type AttackBossPayload = z.infer<typeof AttackBossPayloadSchema>;
export type PlayerPositionPayload = z.infer<typeof PlayerPositionPayloadSchema>;
export type ReconnectWithTokenPayload = z.infer<typeof ReconnectWithTokenPayloadSchema>;

// ============================================
// Validation Helpers
// ============================================

export function validatePayload<T>(
  schema: z.ZodSchema<T>,
  data: unknown
): { success: true; data: T } | { success: false; error: z.ZodError } {
  const result = schema.safeParse(data);
  if (result.success) {
    return { success: true, data: result.data };
  }
  return { success: false, error: result.error };
}
```
