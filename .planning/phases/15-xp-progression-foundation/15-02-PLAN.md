  <action>
    Create client/src/lib/stores/useProgression.tsx:

    ```typescript
    import { create } from 'zustand';
    import { subscribeWithSelector } from 'zustand/middleware';

    export type XPSource = 'vote' | 'boss_damage' | 'consensus' | 'revival';

    interface PendingXPGain {
      id: string;
      amount: number;
      source: XPSource;
      timestamp: number;
    }

    interface LevelUpState {
      active: boolean;
      oldLevel: number;
      newLevel: number;
      triggeredAt: number;
    }

    interface ProgressionState {
      // State
      currentXP: number;
      currentLevel: number;
      pendingXPGains: PendingXPGain[];
      levelUp: LevelUpState | null;
      isLoaded: boolean;

      // Computed helpers
      getProgressToNextLevel: () => { current: number; needed: number; percentage: number };

      // Actions
      handleXPAwarded: (data: {
        playerId: string;
        amount: number;
        source: XPSource;
        newTotal: number;
      }) => void;
      handleLevelUp: (data: {
        playerId: string;
        oldLevel: number;
        newLevel: number;
      }) => void;
      handleSync: (data: {
        playerId: string;
        totalXP: number;
        currentLevel: number;
      }) => void;
      clearPendingGain: (id: string) => void;
      clearLevelUp: () => void;
      reset: () => void;
    }
    ```

    Implementation details:
    - Use subscribeWithSelector middleware (same as useGameState)
    - XP curve formula mirrored client-side for progress calculation:
      - baseXP = 100, exponent = 1.5
      - getLevelThreshold(level) = level <= 1 ? 0 : Math.floor(100 * Math.pow(level - 1, 1.5))
    - getProgressToNextLevel computes current level XP, XP needed for next level, percentage
    - handleXPAwarded: updates currentXP, adds to pendingXPGains with unique id
    - handleLevelUp: sets levelUp state with active=true
    - handleSync: sets currentXP and currentLevel from server (initial load/reconnect)
    - clearPendingGain: removes from pendingXPGains after animation completes
    - clearLevelUp: sets levelUp to null after celebration completes
    - reset: clears all state (on logout/disconnect)

    Update client/src/lib/stores/index.ts if it exists, or create barrel export:

    ```

    import { describe, it, expect, beforeEach } from 'vitest';
    import { useProgression } from './useProgression';

    describe('useProgression', () => {
      beforeEach(() => {
        useProgression.getState().reset();
      });

      describe('handleXPAwarded', () => {
        it('updates currentXP to newTotal', () => {
          useProgression.getState().handleXPAwarded({
            playerId: 'p1',
            amount: 10,
            source: 'vote',
            newTotal: 110,
          });
          expect(useProgression.getState().currentXP).toBe(110);
        });

        it('adds pending gain for animation', () => {
          useProgression.getState().handleXPAwarded({
            playerId: 'p1',
            amount: 10,
            source: 'vote',
            newTotal: 10,
          });
          const pending = useProgression.getState().pendingXPGains;
          expect(pending.length).toBe(1);
          expect(pending[0].amount).toBe(10);
          expect(pending[0].source).toBe('vote');
        });
      });

      describe('handleLevelUp', () => {
        it('sets levelUp state', () => {
          useProgression.getState().handleLevelUp({
            playerId: 'p1',
            oldLevel: 1,
            newLevel: 2,
          });
          const levelUp = useProgression.getState().levelUp;
          expect(levelUp?.active).toBe(true);
          expect(levelUp?.newLevel).toBe(2);
        });
      });

      describe('getProgressToNextLevel', () => {
        it('calculates percentage correctly at level 1', () => {
          useProgression.getState().handleSync({
            playerId: 'p1',
            totalXP: 50,
            currentLevel: 1,
          });
          const progress = useProgression.getState().getProgressToNextLevel();
          expect(progress.current).toBe(50);
          expect(progress.needed).toBe(100);
          expect(progress.percentage).toBe(50);
        });
      });

      describe('clearPendingGain', () => {
        it('removes specific gain by id', () => {
          useProgression.getState().handleXPAwarded({
            playerId: 'p1',
            amount: 10,
            source: 'vote',
            newTotal: 10,
          });
          const id = useProgression.getState().pendingXPGains[0].id;
          useProgression.getState().clearPendingGain(id);
          expect(useProgression.getState().pendingXPGains.length).toBe(0);
        });
      });
    });
    ```
