    describe('ProgressionManager', () => {
      let manager: ProgressionManager;
      let mockEventBus: MockEventBus;

      it('awards XP and emits xp_awarded event', () => {
        manager.awardXP('lobby1', 'player1', 10, 'vote');
        expect(mockEventBus.emitted['progression:xp_awarded']).toBeDefined();
        expect(mockEventBus.emitted['progression:xp_awarded'][0].amount).toBe(10);
      });

      it('emits level_up when crossing threshold', () => {
        manager.setPlayerXP('lobby1', 'player1', 95); // Just under level 2
        manager.awardXP('lobby1', 'player1', 10, 'vote'); // Crosses to 105
        expect(mockEventBus.emitted['progression:level_up']).toBeDefined();
        expect(mockEventBus.emitted['progression:level_up'][0].newLevel).toBe(2);
      });

      it('does not emit level_up when staying same level', () => {
        manager.awardXP('lobby1', 'player1', 5, 'vote');
        expect(mockEventBus.emitted['progression:level_up']).toBeUndefined();
      });

      it('calculates damage XP based on damage amount', () => {
        manager.awardXP('lobby1', 'player1', 10, 'boss_damage');
        // 10 damage * 2 XP/damage = 20 XP
        expect(mockEventBus.emitted['progression:xp_awarded'][0].amount).toBe(20);

    });
