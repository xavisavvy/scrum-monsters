  <action>
    Update ProgressionManager to subscribe to XP-awarding events:

    ```typescript
    constructor(deps: ProgressionManagerDeps) {
      this.eventBus = deps.eventBus;
      this.storage = deps.storage;

      // XP-01: Vote submission
      this.eventBus.on('estimation:vote_cast', this.handleVoteCast.bind(this));

      // XP-02: Boss damage
      this.eventBus.on('combat:boss_damaged', this.handleBossDamaged.bind(this));

      // XP-03: Consensus bonus
      this.eventBus.on('estimation:full_consensus_reached', this.handleConsensus.bind(this));

      // XP-04: Revival
      this.eventBus.on('combat:player_revived', this.handleRevival.bind(this));
    }

    private handleVoteCast(payload: EstimationVoteCastPayload): void {
      // Award 10 XP for voting
      this.awardXP(payload.lobbyId, payload.playerId, XP_RATES.vote, 'vote');
    }

    private handleBossDamaged(payload: CombatBossDamagedPayload): void {
      // Award 2 XP per damage point
      const xp = payload.damage * XP_RATES.boss_damage_multiplier;
      this.awardXP(payload.lobbyId, payload.playerId, xp, 'boss_damage');
    }

    private handleConsensus(payload: EstimationFullConsensusReachedPayload): void {
      // Award 50 XP to all players who voted
      const lobby = this.getLobbyPlayers(payload.lobbyId);
      lobby.forEach(playerId => {
        this.awardXP(payload.lobbyId, playerId, XP_RATES.consensus, 'consensus');
      });
    }

    private handleRevival(payload: CombatPlayerRevivedPayload): void {
      // Award 30 XP to reviver
      this.awardXP(payload.lobbyId, payload.reviverId, XP_RATES.revival, 'revival');
    }
    ```

    - revival: 30

    ```typescript
    socket.on('progression:xp_awarded', (data) => {
      const { currentPlayer } = useGameState.getState();
      if (data.playerId === currentPlayer?.id) {
        useProgression.getState().handleXPAwarded({
          playerId: data.playerId,
          amount: data.amount,
          source: data.source,
          newTotal: data.newTotal,
        });
      }
    });

    socket.on('progression:level_up', (data) => {
      const { currentPlayer } = useGameState.getState();
      if (data.playerId === currentPlayer?.id) {
        useProgression.getState().handleLevelUp({
          playerId: data.playerId,
          oldLevel: data.oldLevel,
          newLevel: data.newLevel,
        });
      }
    });

    socket.on('progression:sync', (data) => {
      const { currentPlayer } = useGameState.getState();
      if (data.playerId === currentPlayer?.id) {
        useProgression.getState().handleSync({
          playerId: data.playerId,
          totalXP: data.totalXP,
          currentLevel: data.currentLevel,
        });
      }
    });
    ```

    Note: Only process events for the current player. Other players' XP is not shown (per CONTEXT.md - levels visible in lobby only).
