Error handling pattern:
```typescript
socket.on('join_lobby', ({ lobbyId, playerName }) => {
  try {
    const { lobby, player } = sessionManager.joinLobby(lobbyId, playerName);
    // ... existing emit logic
  } catch (error) {
    if (error instanceof LobbyNotFoundError) {
      socket.emit('game_error', { message: 'Lobby not found' });
    } else if (error instanceof SessionError) {
      socket.emit('game_error', { message: error.message });
    } else {
      console.error('Unexpected error:', error);
      socket.emit('game_error', { message: 'An unexpected error occurred' });
    }
  }
});
```

Reconnect handler pattern:
```typescript
socket.on('reconnect_with_token', ({ reconnectToken }) => {
  const result = sessionManager.attemptPlayerReconnect(reconnectToken);

  if (result.result === 'success' && result.lobbySync) {
    socket.data.playerId = result.lobbySync.yourPlayer.id;
    socket.data.lobbyId = result.lobbySync.lobby.id;
    socket.join(result.lobbySync.lobby.id);
    socket.emit('lobby_sync', result.lobbySync);
    socket.to(result.lobbySync.lobby.id).emit('player_reconnected', {
      player: result.lobbySync.yourPlayer
    });
    console.log(`Reconnected: ${result.lobbySync.yourPlayer.name}`);
  } else {
    socket.emit('reconnect_failed', { reason: result.result, message: result.message });
  }
});
```

IMPORTANT: Keep backward compatibility:
