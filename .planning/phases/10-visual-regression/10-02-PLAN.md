    if (await createButton.isVisible().catch(() => false)) {
      const nameInput = page.getByPlaceholder(/name/i);

      await waitForStableUI(page);

        await expect(page).toHaveScreenshot('voting-screen-desktop.png', {
          ...screenshotOptions(page),
